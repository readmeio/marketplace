import React, { useState, Children, isValidElement, cloneElement } from 'react';

export const Steps = ({ name = 'steps', children }) => {
  const childArray = Children.toArray(children);
  const count = childArray.length;
  
  return (
    <div className="steps-container" data-steps-name={name}>
      {childArray.map((child, index) => {
        if (isValidElement(child)) {
          return cloneElement(child, {
            key: index,
            stepIndex: index,
            stepsName: name,
            isFirst: index === 0,
            isLast: index === count - 1,
          });
        }
        return child;
      })}
    </div>
  );
};

export const Step = ({
  number,
  stepIndex = 0,
  title = '',
  stepsName = 'steps',
  isFirst = false,
  isLast = false,
  children,
}) => {
  const [isHovered, setIsHovered] = useState(false);
  const [copied, setCopied] = useState(false);
  const displayNumber = number !== undefined ? number : stepIndex + 1;
  const stepId = `${stepsName}-${displayNumber}`;

  const copyLink = () => {
    if (typeof window === 'undefined') return;
    
    try {
      const url = `${window.location.origin}${window.location.pathname}#${stepId}`;
      navigator.clipboard.writeText(url);
      setCopied(true);
      setTimeout(() => setCopied(false), 1500);
    } catch (error) {
      console.error('Failed to copy link:', error);
    }
  };

  return (
    <div 
      className="step relative" 
      style={{ paddingLeft: '44px' }}
    >
      {/* Vertical line - positioned behind the number, extends into padding */}
      <div
        className="absolute bg-gray-100 dark:bg-gray-800"
        style={{
          width: '1px',
          left: '13.5px',
          top: isFirst ? '28px' : '-24px',
          bottom: isLast ? '0' : '-24px',
        }}
      />
      
      {/* Number button - positioned on top of line */}
      <button
        onClick={copyLink}
        onMouseEnter={() => setIsHovered(true)}
        onMouseLeave={() => setIsHovered(false)}
        className={`
          absolute left-0 top-0
          w-7 h-7 rounded-lg
          border
          flex items-center justify-center
          text-gray-500 dark:text-gray-400
          font-normal text-base
          cursor-pointer
          transition-all duration-200
          z-10
          ${copied 
            ? 'bg-gray-200 dark:bg-gray-600 border-gray-400 dark:border-gray-500' 
            : 'bg-white dark:bg-gray-900 border-gray-200 dark:border-gray-700 hover:border-gray-300 dark:hover:border-gray-600 hover:bg-gray-50 dark:hover:bg-gray-800'
          }
        `}
        title="Copy link to this step"
      >
        {copied ? (
          <i className="fa-solid fa-check text-black text-s" aria-label="Copied" />
        ) : isHovered ? (
          <i className="fa-solid fa-link text-xs" aria-label="Copy link" />
        ) : (
          displayNumber
        )}
      </button>
      
      {/* Right column: title + content */}
      <div className={isLast ? '' : 'pb-6'}>
        <h3 
          id={stepId}
          className="text-base font-semibold text-gray-900 dark:text-white m-0"
          style={{ lineHeight: '28px' }}
        >
          {title}
        </h3>
        <div className="text-gray-600 dark:text-gray-300 text-sm mt-2">
          {children}
        </div>
      </div>
    </div>
  );
};
